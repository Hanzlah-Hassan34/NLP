# DentaBot Evaluation - Quick Run Commands

# 1. Start the API server.
python -m uvicorn app.main:app --port 8000

# 2. Run focused tool tests.
python -m pytest evals/test_tools_crm.py evals/test_tools_appointment.py evals/test_tools_weather.py evals/test_tools_cost.py -v

# 3. Run the quick RAG smoke evaluation.
python quick_eval.py

# 4. Run Locust load testing.
locust -f evals/locust_ws.py --host=http://localhost:8000 --headless -u 5 -r 1 -t 30s DentaBotHttpUser

# For the Locust web UI, omit the headless flags and open http://localhost:8089.
locust -f evals/locust_ws.py --host=http://localhost:8000
